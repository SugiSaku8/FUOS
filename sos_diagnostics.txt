struct Camera {
    view_proj: mat4x4<f32>,
};

@group(0)
@binding(0)
var<uniform> camera: Camera;

struct VertexInput {
    @location(0) position: vec3<f32>,
    @location(1) height: f32,
};

struct VertexOutput {
    @builtin(position) position: vec4<f32>,
    @location(0) height: f32,
};

@vertex
fn vs_main(input: VertexInput) -> VertexOutput {
    var out: VertexOutput;
    out.position = camera.view_proj * vec4<f32>(input.position, 1.0);
    out.height = input.height;
    return out;
}

@fragment
fn fs_main(input: VertexOutput) -> @location(0) vec4<f32> {
    let h = input.height;

    // 海面：透明感のある青色
    if h <= 0.0 {
        return vec4<f32>(0.025, 0.32, 0.58, 0.28);
    }

    // 陸地
    let land_t = clamp(h / 1000.0, 0.0, 1.0);
    let lowland = vec3<f32>(0.12, 0.25, 0.12);
    let highland = vec3<f32>(0.28, 0.30, 0.22);
    let land_color = mix(lowland, highland, land_t);

    return vec4<f32>(land_color, 1.0);
}

use std::fs::File;

fn load_bathymetry(
    path: &str,
    src_width: usize,
    src_height: usize,
    dst_width: usize,
    dst_height: usize,
) -> anyhow::Result<Vec<f32>> {
    let mut file = File::open(path)?;

    let mut bytes = Vec::new();

    file.read_to_end(&mut bytes)?;

    let src = bytes
        .chunks_exact(4)
        .map(|b| f32::from_le_bytes([b[0], b[1], b[2], b[3]]))
        .collect::<Vec<_>>();

    if src.len() != src_width * src_height {
        anyhow::bail!(
            "source bathymetry size mismatch: {} != {}",
            src.len(),
            src_width * src_height
        );
    }

    let mut dst = vec![0.0; dst_width * dst_height];

    for y in 0..dst_height {
        let sy = y * src_height / dst_height;

        for x in 0..dst_width {
            let sx = x * src_width / dst_width;

            dst[y * dst_width + x] = src[sy * src_width + sx];
        }
    }

    Ok(dst)
}

//! SOS dataset builder.

use std::path::Path;

use anyhow::Result;

use s_field::{
    BoundaryField, Coordinates, Metadata, OceanField, SalinityField, ScalarField, TemperatureField,
    VelocityField,
};

use s_sof::builder::SofPackageBuilder;

use crate::hycom::HycomDataset;

/// Builder for Suruga Ocean Simulator dataset.
pub struct SurugaBuilder;

impl SurugaBuilder {
    /// Build SOF package from HYCOM dataset.
    pub fn build<P: AsRef<Path>>(
        output: P,
        dataset: HycomDataset,
        latitude: Vec<f64>,
        longitude: Vec<f64>,
        depth: Vec<f32>,
        time: Vec<f64>,
        bathymetry: Vec<f32>,
    ) -> Result<()> {
        let grid = dataset.grid.clone();

        //
        // Coordinates
        //

        let coordinates = Coordinates::new(
            grid.clone(),
            latitude.clone(),
            longitude.clone(),
            depth.clone(),
            time.clone(),
        );

        //
        // Metadata
        //

        let mut metadata = Metadata::new();

        metadata.set_dataset_name("Suruga Bay");

        metadata.set_title("Suruga Ocean Simulator HYCOM Dataset");

        //
        // Temperature
        //

        let temperature = TemperatureField::from_field(dataset.temperature.into_field4d());

        //
        // Salinity
        //

        let salinity = SalinityField::from_field(dataset.salinity.into_field4d());

        //
        // Velocity
        //

        let velocity = VelocityField::from_fields(
            grid.clone(),
            dataset.u,
            dataset.v,
            ScalarField::zeros(grid.clone()),
        );

        //
        // OceanField
        //

        let boundary = BoundaryField::zeros(velocity.grid().clone());

        let ocean = OceanField::new(
            coordinates,
            metadata,
            velocity,
            temperature,
            salinity,
            boundary,
        );
        //
        // Surface forcing
        //
        // MVP:
        // wind_u
        // wind_v
        // surface_heat
        //

        let size = grid.len();

        let wind_u = vec![0.0f32; size];

        let wind_v = vec![0.0f32; size];

        let surface_heat = vec![0.0f32; size];

        //
        // Grid coordinates
        //

        let latitude32 = latitude.iter().map(|v| *v as f32).collect();

        let longitude32 = longitude.iter().map(|v| *v as f32).collect();

        //
        // Build SOF
        //

        SofPackageBuilder::new("suruga")
            //
            // grid.snc
            //
            .grid(grid.clone(), latitude32, longitude32, depth, bathymetry)
            //
            // boundary.snc
            //
            .boundary(ocean.clone())
            //
            // initial.snc
            //
            .initial(ocean.clone())
            //
            // restart.snc
            //
            .restart(ocean.clone())
            //
            // forcing.snc
            //
            .forcing(wind_u, wind_v, surface_heat)
            .build(output)?;

        Ok(())
    }
}

----- crates/s-field/src/generator/kuroshio.rs -----
use crate::{Grid4D, VelocityField};

/// 駿河湾へ流入する黒潮の簡易モデルを生成する。
///
/// 南側境界から北東方向へ流入する。
pub fn create_kuroshio(grid: Grid4D) -> VelocityField {
    let mut velocity = VelocityField::new(grid);

    let shape = velocity.shape();

    let nx = shape[0];
    let ny = shape[1];
    let nz = shape[2];

    //
    // 値の書き込み
    //
    {
        let u = velocity.u_mut().time_slice_mut(0).unwrap();

        for z in 0..nz {
            //
            // 表層ほど強い
            //
            let depth_factor = 1.0 - z as f32 / nz as f32;

            for y in 0..ny {
                for x in 0..nx {
                    let index = z * nx * ny + y * nx + x;

                    //
                    // 南側入口
                    //
                    let south = y as f32 / ny as f32;

                    u[index] = 0.8 * depth_factor * (1.0 - south);
                }
            }
        }
    }

    {
        let v = velocity.v_mut().time_slice_mut(0).unwrap();

        for z in 0..nz {
            let depth_factor = 1.0 - z as f32 / nz as f32;

            for y in 0..ny {
                for x in 0..nx {
                    let index = z * nx * ny + y * nx + x;

                    //
                    // 少し北向き
                    //
                    v[index] = 0.3 * depth_factor;
                }
            }
        }
    }

    {
        let w = velocity.w_mut().time_slice_mut(0).unwrap();

        for value in w.iter_mut() {
            *value = 0.0;
        }
    }

    velocity
}

//! Velocity field.

use crate::{Coordinates, Grid4D, ScalarField};

/// 3次元流速場.
///
/// Components:
///
/// u : east-west velocity
/// v : north-south velocity
/// w : vertical velocity
///
#[derive(Clone, Debug)]
pub struct VelocityField {
    grid: Grid4D,

    coordinates: Option<Coordinates>,

    u: ScalarField<f32>,

    v: ScalarField<f32>,

    w: ScalarField<f32>,
}

impl VelocityField {
    /// Creates empty velocity field.
    pub fn new(grid: Grid4D) -> Self {
        Self {
            u: ScalarField::zeros(grid.clone()),

            v: ScalarField::zeros(grid.clone()),

            w: ScalarField::zeros(grid.clone()),

            coordinates: None,

            grid,
        }
    }

    /// Creates velocity field with coordinates.
    pub fn with_coordinates(grid: Grid4D, coordinates: Coordinates) -> Self {
        Self {
            u: ScalarField::zeros(grid.clone()),

            v: ScalarField::zeros(grid.clone()),

            w: ScalarField::zeros(grid.clone()),

            coordinates: Some(coordinates),

            grid,
        }
    }

    #[inline]
    pub fn grid(&self) -> &Grid4D {
        &self.grid
    }

    #[inline]
    pub fn shape(&self) -> [usize; 3] {
        [self.grid.depth(), self.grid.height(), self.grid.width()]
    }

    /// Returns grid spacing.

    #[inline]

    pub fn spacing(&self) -> [f32; 4] {
        self.grid.spacing()
    }

    #[inline]
    pub fn coordinates(&self) -> Option<&Coordinates> {
        self.coordinates.as_ref()
    }

    #[inline]
    pub fn coordinates_mut(&mut self) -> Option<&mut Coordinates> {
        self.coordinates.as_mut()
    }

    #[inline]
    pub fn set_coordinates(&mut self, coordinates: Coordinates) {
        self.coordinates = Some(coordinates);
    }

    #[inline]
    pub fn u(&self) -> &ScalarField<f32> {
        &self.u
    }

    #[inline]
    pub fn v(&self) -> &ScalarField<f32> {
        &self.v
    }

    #[inline]
    pub fn w(&self) -> &ScalarField<f32> {
        &self.w
    }

    #[inline]
    pub fn u_mut(&mut self) -> &mut ScalarField<f32> {
        &mut self.u
    }

    #[inline]
    pub fn v_mut(&mut self) -> &mut ScalarField<f32> {
        &mut self.v
    }

    #[inline]
    pub fn w_mut(&mut self) -> &mut ScalarField<f32> {
        &mut self.w
    }

    /// Returns velocity by linear index.
    pub fn velocity_at_index(&self, time: usize, index: usize) -> Option<[f32; 3]> {
        Some([
            *self.u.time_slice(time)?.get(index)?,
            *self.v.time_slice(time)?.get(index)?,
            *self.w.time_slice(time)?.get(index)?,
        ])
    }

    /// Returns velocity by grid index.
    pub fn velocity(&self, time: usize, x: usize, y: usize, z: usize) -> Option<[f32; 3]> {
        if x >= self.grid.width() || y >= self.grid.height() || z >= self.grid.depth() {
            return None;
        }

        let index = z * self.grid.width() * self.grid.height() + y * self.grid.width() + x;

        self.velocity_at_index(time, index)
    }

    /// Samples velocity from physical coordinates.
    ///
    /// Coordinates:
    ///
    /// longitude
    /// latitude
    /// depth
    ///
    pub fn sample(&self, longitude: f64, latitude: f64, depth: f32, time: f64) -> Option<[f32; 3]> {
        let coord = self.coordinates.as_ref()?;

        let x = coord.nearest_longitude(longitude);

        let y = coord.nearest_latitude(latitude);

        let z = coord.nearest_depth(depth);

        let t = coord.nearest_time(time);

        self.velocity(t, x, y, z)
    }
    /// Samples velocity using 4D multilinear interpolation.
    ///
    /// Interpolates across longitude, latitude, depth, and time.
    ///
    /// Returns None when:
    /// - Spatial coordinates are outside the dataset.
    /// - The requested time is outside the available time range.
    /// - Coordinates are invalid or non-monotonic.
    /// - A required velocity value is non-finite.
    pub fn sample_interpolated(
        &self,
        longitude: f64,
        latitude: f64,
        depth: f32,
        time: f64,
    ) -> Option<[f32; 3]> {
        let coord = self.coordinates.as_ref()?;

        if !longitude.is_finite()
            || !latitude.is_finite()
            || !depth.is_finite()
            || !time.is_finite()
        {
            return None;
        }

        if !coord.contains(longitude, latitude, depth) {
            return None;
        }

        // Find the two surrounding grid indices on each axis.
        let (x0, x1, wx) = bracket_axis(coord.longitude(), longitude)?;

        let (y0, y1, wy) = bracket_axis(coord.latitude(), latitude)?;

        let depth_values: Vec<f64> = coord.depth().iter().map(|&value| value as f64).collect();

        let (z0, z1, wz) = bracket_axis(&depth_values, depth as f64)?;

        // A single time slice represents a time-independent field.
        let (t0, t1, wt) = if coord.time_steps() == 1 {
            (0, 0, 0.0)
        } else {
            bracket_axis(coord.time(), time)?
        };

        let xs = [(x0, 1.0 - wx), (x1, wx)];
        let ys = [(y0, 1.0 - wy), (y1, wy)];
        let zs = [(z0, 1.0 - wz), (z1, wz)];
        let ts = [(t0, 1.0 - wt), (t1, wt)];

        let mut result = [0.0_f64; 3];

        for &(t, tw) in &ts {
            for &(z, zw) in &zs {
                for &(y, yw) in &ys {
                    for &(x, xw) in &xs {
                        let weight = tw * zw * yw * xw;

                        // Avoid reading unnecessary corners when a query
                        // lies exactly on a grid coordinate.
                        if weight == 0.0 {
                            continue;
                        }

                        let velocity = self.velocity(t, x, y, z)?;

                        for component in 0..3 {
                            let value = velocity[component];

                            if !value.is_finite() {
                                return None;
                            }

                            result[component] += value as f64 * weight;
                        }
                    }
                }
            }
        }

        let interpolated = [result[0] as f32, result[1] as f32, result[2] as f32];

        if interpolated.iter().all(|value| value.is_finite()) {
            Some(interpolated)
        } else {
            None
        }
    }

    /// Clears all velocity values.
    pub fn clear(&mut self) {
        self.u.fill(0.0);

        self.v.fill(0.0);

        self.w.fill(0.0);
    }
    /// Creates velocity field from existing scalar fields.
    pub fn from_fields(
        grid: Grid4D,
        u: ScalarField<f32>,
        v: ScalarField<f32>,
        w: ScalarField<f32>,
    ) -> Self {
        Self {
            grid,
            coordinates: None,
            u,
            v,
            w,
        }
    }
}

use s_metal::{

fn index3d(x: usize, y: usize, z: usize, nx: usize, ny: usize) -> usize {
    z * nx * ny + y * nx + x
}

/// 内部セルの発散 RMS を計算する。
fn rms_interior(values: &[f32], nx: usize, ny: usize, nz: usize) -> f64 {
    if nx < 3 || ny < 3 || nz < 3 {
        return 0.0;
    }

    let mut sum_squared = 0.0_f64;
    let mut count = 0_usize;
    for z in 1..nz - 1 {
        for y in 1..ny - 1 {
            for x in 1..nx - 1 {
                let i = index3d(x, y, z, nx, ny);
                let value = values[i] as f64;
                sum_squared += value * value;
                count += 1;
            }
        }
    }
    if count == 0 {
        return 0.0;
    }
    (sum_squared / count as f64).sqrt()
}

/// 圧力方程式の相対残差を計算する。
///
/// 対象方程式:
///     laplacian(p) = (rho / dt) * divergence
///
/// 戻り値:
///     (残差 RMS, 相対残差)
fn pressure_residual_rms(
    pressure: &[f32],
    divergence: &[f32],
    nx: usize,
    ny: usize,
    nz: usize,
    dx: f32,
    dy: f32,
    dz: f32,
    dt: f32,
    rho: f32,
) -> (f64, f64) {
    if nx < 3 || ny < 3 || nz < 3 {
        return (0.0, 0.0);
    }

    let dx = dx as f64;
    let dy = dy as f64;
    let dz = dz as f64;
    let wx = 1.0 / (dx * dx);
    let wy = 1.0 / (dy * dy);
    let wz = 1.0 / (dz * dz);
    let scale = rho as f64 / dt as f64;
    let mut residual_squared_sum = 0.0_f64;
    let mut rhs_squared_sum = 0.0_f64;
    let mut count = 0_usize;
    for z in 1..nz - 1 {
        for y in 1..ny - 1 {
            for x in 1..nx - 1 {
                let i = index3d(x, y, z, nx, ny);
                let xm = index3d(x - 1, y, z, nx, ny);
                let xp = index3d(x + 1, y, z, nx, ny);
                let ym = index3d(x, y - 1, z, nx, ny);
                let yp = index3d(x, y + 1, z, nx, ny);
                let zm = index3d(x, y, z - 1, nx, ny);
                let zp = index3d(x, y, z + 1, nx, ny);
                let p = pressure[i] as f64;
                let laplacian = wx * (pressure[xm] as f64 - 2.0 * p + pressure[xp] as f64)
                    + wy * (pressure[ym] as f64 - 2.0 * p + pressure[yp] as f64)
                    + wz * (pressure[zm] as f64 - 2.0 * p + pressure[zp] as f64);
                let rhs = scale * divergence[i] as f64;
                let residual = laplacian - rhs;
                residual_squared_sum += residual * residual;
                rhs_squared_sum += rhs * rhs;
                count += 1;
            }
        }
    }
    if count == 0 {
        return (0.0, 0.0);
    }
    let residual_rms = (residual_squared_sum / count as f64).sqrt();
    let rhs_rms = (rhs_squared_sum / count as f64).sqrt();
    let relative_residual = if rhs_rms > 1.0e-30 {
        residual_rms / rhs_rms
    } else {
        0.0
    };
    (residual_rms, relative_residual)
}

struct ProjectionResult {
    iterations: usize,
    divergence_before: f64,
    divergence_after: f64,
    reduction_percent: f64,
    residual_rms: f64,
    relative_residual: f64,
    elapsed_ms: f64,
}

/// 速度場に圧力補正を適用し、結果を測定する。
fn run_projection_case(
    context: &MetalContext,

    divergence_operator: &DivergenceOperator,

    pressure_solver: &PressureSolver,

    gradient_operator: &GradientPressureOperator,

    nx: usize,

    ny: usize,

    nz: usize,

    dt: f32,

    rho: f32,

    iterations: usize,
) -> ProjectionResult {
    let dx = 1.0_f32;
    let dy = 1.0_f32;
    let dz = 1.0_f32;

    let grid = GridInfo::new(nx as u32, ny as u32, nz as u32, dx, dy, dz);
    let velocity = make_test_velocity(context, grid, nx, ny, nz);
    let start = Instant::now();
    // 1. 補正前の発散
    let divergence_before_field = divergence_operator.compute(context, &velocity);
    let divergence_before_values = divergence_before_field.download(context);
    let divergence_before = rms_interior(&divergence_before_values, nx, ny, nz);
    // 2. 圧力ポアソン方程式を解く
    let pressure = pressure_solver.solve(context, &divergence_before_field, dt, rho, iterations);
    let pressure_values = pressure.download(context);
    // 3. 圧力方程式の残差
    let (residual_rms, relative_residual) = pressure_residual_rms(
        &pressure_values,
        &divergence_before_values,
        nx,
        ny,
        nz,
        dx,
        dy,
        dz,
        dt,
        rho,
    );
    // 4. 圧力勾配による速度補正
    let corrected_velocity = gradient_operator.compute(context, &velocity, &pressure, dt, rho);
    // 5. 補正後の発散
    let divergence_after_field = divergence_operator.compute(context, &corrected_velocity);
    let divergence_after_values = divergence_after_field.download(context);
    let divergence_after = rms_interior(&divergence_after_values, nx, ny, nz);
    let reduction_percent = if divergence_before > 1.0e-30 {
        (1.0 - divergence_after / divergence_before) * 100.0
    } else {
        0.0
    };
    let elapsed_ms = start.elapsed().as_secs_f64() * 1000.0;
    ProjectionResult {
        iterations,
        divergence_before,
        divergence_after,
        reduction_percent,
        residual_rms,
        relative_residual,
        elapsed_ms,
    }
}

    // Coordinates
    //
    let coordinates =
        Coordinates::new(grid, vec![0.0; nx], vec![0.0; ny], vec![0.0; nz], vec![0.0]);

    //
    // Metadata
    //
    let metadata = Metadata::default();

    //
    // Ocean fields
    //
    let velocity = VelocityField::new(grid);

    let temperature = TemperatureField::new(grid);

    let salinity = SalinityField::new(grid);

    //
    // OceanField
    //
    let mut ocean = OceanField::new(coordinates, metadata, velocity, temperature, salinity);

    //
    // Initial velocity
    //
    let cx = nx / 2;
    let cy = ny / 2;
    let cz = nz / 2;

    for z in 0..nz {
        for y in 0..ny {
            for x in 0..nx {
                let dx = x as f32 - cx as f32;

                let dy = y as f32 - cy as f32;

                let dz = z as f32 - cz as f32;

                let r2 = dx * dx + dy * dy + dz * dz;

                let value = (-r2 / 20.0).exp();

                let i = z * nx * ny + y * nx + x;

                //
                // u velocity
                //
                {
                    let u = ocean.velocity_mut().u_mut().time_slice_mut(0).unwrap();

                    u[i] = value;
                }

                //
                // v velocity
                //
                {
                    let v = ocean.velocity_mut().v_mut().time_slice_mut(0).unwrap();

                    v[i] = 0.0;
                }

                //
                // w velocity
                //
                {
                    let w = ocean.velocity_mut().w_mut().time_slice_mut(0).unwrap();

                    w[i] = 0.0;
                }
            }
        }
    }

    //
    // CPU initial check
    //
    let before = ocean.velocity().u().time_slice(0).unwrap();

    println!(
        "CPU before center = {}",
        before[cz * nx * ny + cy * nx + cx]
    );

    //
    // Solver
    //
    let solver = OceanSolver::new(context);

    solver.step(&mut ocean, 0.01);

    //
    // CPU result
    //
    let after = ocean.velocity().u().time_slice(0).unwrap();

    println!("CPU after center = {}", after[cz * nx * ny + cy * nx + cx]);

    //
    // Neighborhood
    //
    println!("neighborhood:");

    for z in cz - 1..=cz + 1 {
        for y in cy - 1..=cy + 1 {
            for x in cx - 1..=cx + 1 {
                let i = z * nx * ny + y * nx + x;

                println!("{} {} {} : {}", x, y, z, after[i]);
            }
        }
    }
}

use crate::{GpuVelocityField, MetalContext};

use s_field::{Grid4D, VelocityField};

/// Ocean velocity boundary conditions.
///
/// MVP:
/// - Prescribes velocity on the x = nx - 1 boundary face.
/// - Preserves the velocity values in the interior.
///
/// This is a test boundary condition, not a validated Kuroshio model.
pub struct BoundaryOperator;

impl BoundaryOperator {
    pub fn new(_context: &MetalContext) -> Self {
        Self
    }
    /// Construct a test inflow velocity field.
    pub fn kuroshio_inflow(
        &self,
        context: &MetalContext,
        velocity: &GpuVelocityField,
    ) -> GpuVelocityField {
        let grid_info = velocity.grid();

        let nx = grid_info.nx() as usize;
        let ny = grid_info.ny() as usize;
        let nz = grid_info.nz() as usize;

        let total = nx * ny * nz;

        let grid = Grid4D::new(
            [nx, ny, nz, 1],
            [grid_info.dx(), grid_info.dy(), grid_info.dz(), 1.0],
        );

        let mut field = VelocityField::new(grid);

        let mut u = vec![0.0_f32; total];
        let mut v = vec![0.0_f32; total];
        let w = vec![0.0_f32; total];

        if nx > 0 && ny > 0 && nz > 0 {
            for z in 0..nz {
                let depth_factor = if nz > 1 {
                    1.0 - z as f32 / (nz - 1) as f32
                } else {
                    1.0
                };

                for y in 0..ny {
                    let index = z * nx * ny + y * nx + (nx - 1);

                    u[index] = 0.8 * depth_factor;
                    v[index] = 0.15 * depth_factor;
                }
            }
        }

        field.u_mut().replace(u);
        field.v_mut().replace(v);
        field.w_mut().replace(w);

        GpuVelocityField::from_velocity(context, &field)
    }

    /// Apply inflow on one boundary face while preserving the interior.
    ///
    /// This MVP implementation transfers the field through CPU memory.
    pub fn apply_velocity(&self, context: &MetalContext, velocity: &mut GpuVelocityField) {
        let grid_info = velocity.grid();
        let nx = grid_info.nx() as usize;
        let ny = grid_info.ny() as usize;
        let nz = grid_info.nz() as usize;

        if nx == 0 || ny == 0 || nz == 0 {
            return;
        }

        // 現在の速度場をCPUへダウンロード
        let mut current = velocity.download(context);

        // 流入境界の速度場を作成
        let inflow = self.kuroshio_inflow(context, velocity);
        let inflow_cpu = inflow.download(context);

        // u, v, w をそれぞれ独立した配列として取得
        let mut u = current
            .u()
            .time_slice(0)
            .expect("Failed to access current u velocity")
            .to_vec();

        let mut v = current
            .v()
            .time_slice(0)
            .expect("Failed to access current v velocity")
            .to_vec();

        let mut w = current
            .w()
            .time_slice(0)
            .expect("Failed to access current w velocity")
            .to_vec();

        let inflow_u = inflow_cpu
            .u()
            .time_slice(0)
            .expect("Failed to access inflow u velocity");

        let inflow_v = inflow_cpu
            .v()
            .time_slice(0)
            .expect("Failed to access inflow v velocity");

        let inflow_w = inflow_cpu
            .w()
            .time_slice(0)
            .expect("Failed to access inflow w velocity");

        // x方向の端面に流入境界条件を適用
        for z in 0..nz {
            for y in 0..ny {
                let index = z * nx * ny + y * nx + (nx - 1);

                u[index] = inflow_u[index];
                v[index] = inflow_v[index];
                w[index] = inflow_w[index];
            }
        }

        // 変更した配列を速度場へ戻す
        current.u_mut().replace(u);
        current.v_mut().replace(v);
        current.w_mut().replace(w);

        // 更新した速度場をGPUへアップロード
        *velocity = GpuVelocityField::from_velocity(context, &current);
    }
}

            width: ((grid.nx() + 7) / 8) as u64,

            height: ((grid.ny() + 7) / 8) as u64,

            depth: ((grid.nz() + 7) / 8) as u64,
        };

        encoder.dispatch_thread_groups(groups, threads);

        encoder.end_encoding();

        command_buffer.commit();

        command_buffer.wait_until_completed();

    }

        &self,
        context: &MetalContext,
        velocity: &GpuVelocityField,

            v,

            w,

            grid: velocity.grid(),
        }
    }

//! Ocean solver using Metal compute.

use crate::{
    AdvectionOperator, BoundaryOperator, DiffusionOperator, DivergenceOperator, GpuOceanState,
    GradientPressureOperator, MetalContext, PressureSolver,
};
use std::sync::Arc;

pub struct OceanSolver {
    context: Arc<MetalContext>,

    boundary: BoundaryOperator,

    advection: AdvectionOperator,

    diffusion: DiffusionOperator,

    divergence: DivergenceOperator,

    pressure: PressureSolver,

    gradient_pressure: GradientPressureOperator,
}
impl OceanSolver {
    pub fn new(context: Arc<MetalContext>) -> Self {
        let boundary = BoundaryOperator::new(&context);

        let advection = AdvectionOperator::new(&context);

        let diffusion = DiffusionOperator::new(&context);

        let divergence = DivergenceOperator::new(&context);

        let pressure = PressureSolver::new(&context);

        let gradient_pressure = GradientPressureOperator::new(&context);

        Self {
            context,

            boundary,

            advection,

            diffusion,

            divergence,

            pressure,

            gradient_pressure,
        }
    }

    /// Advance ocean state.
    ///
    /// Navier-Stokes simplified step.
    ///
    ///
    /// ∂u/∂t
    ///
    /// +
    ///
    /// (u・∇)u
    ///
    /// =
    ///
    /// -1/rho ∇p
    ///
    /// +
    ///
    /// ν∇²u
    ///
    pub fn step(&self, state: &mut GpuOceanState, dt: f32) {
        //
        // 1. Advection
        //
        state.velocity = self.advection.compute(&self.context, &state.velocity, dt);

        //
        // 2. Diffusion
        //
        state.velocity =
            self.diffusion
                .compute_velocity(&self.context, &state.velocity, 0.000001, dt);

        //
        // 3. Boundary condition
        //
        self.boundary
            .apply_velocity(&self.context, &mut state.velocity);

        //
        // 4. Divergence
        //
        // 速度場の発散を計算
        let divergence = self.divergence.compute(&self.context, &state.velocity);

        // 海水密度の代表値
        let rho = 1025.0_f32;

        // 圧力ポアソン方程式を解く
        state.pressure = self.pressure.solve(&self.context, &divergence, dt, rho, 40);

        // 圧力勾配によって速度場を補正
        state.velocity = self.gradient_pressure.compute(
            &self.context,
            &state.velocity,
            &state.pressure,
            dt,
            rho,
        );

        //
        // 7. Boundary correction
        //
        self.boundary
            .apply_velocity(&self.context, &mut state.velocity);
    }
}

//! GPU velocity field.

use crate::{GpuField3D, GridInfo, MetalContext};

use s_field::VelocityField;

/// GPU resident velocity field.
///
/// Components:
///
/// u : east-west velocity
/// v : north-south velocity
/// w : vertical velocity
///
pub struct GpuVelocityField {
    pub u: GpuField3D,

    pub v: GpuField3D,

    pub w: GpuField3D,

    pub(crate) grid: GridInfo,
}

impl GpuVelocityField {
    /// Upload CPU velocity field to GPU.
    pub fn from_velocity(context: &MetalContext, velocity: &VelocityField) -> Self {
        let shape = velocity.shape();

        let spacing = velocity.spacing();

        let grid = GridInfo::new(
            shape[0] as u32,
            shape[1] as u32,
            shape[2] as u32,
            spacing[0],
            spacing[1],
            spacing[2],
        );

        //
        // First time slice
        //
        let u = velocity.u().time_slice(0).unwrap();

        let v = velocity.v().time_slice(0).unwrap();

        let w = velocity.w().time_slice(0).unwrap();

        let gpu = Self {
            u: GpuField3D::from_f32(context, u, grid),

            v: GpuField3D::from_f32(context, v, grid),

            w: GpuField3D::from_f32(context, w, grid),

            grid,
        };

        //
        // Debug upload check
        //
        let center =
            (shape[2] / 2) * shape[0] * shape[1] + (shape[1] / 2) * shape[0] + shape[0] / 2;

        let uploaded = gpu.u.download(context);

        println!("GPU upload center = {}", uploaded[center]);

        gpu
    }

    /// Create empty velocity field.
    ///
    /// Used for compute output.
    pub fn empty(context: &MetalContext, grid: GridInfo) -> Self {
        Self {
            u: GpuField3D::empty(context, grid),

            v: GpuField3D::empty(context, grid),

            w: GpuField3D::empty(context, grid),

            grid,
        }
    }

    /// Download GPU velocity field to CPU.
    ///
    /// Converts GPU resident velocity into s-field VelocityField.

    #[inline]
    pub fn grid(&self) -> GridInfo {
        self.grid
    }

    /// Download GPU velocity field to CPU arrays.
    pub fn download(&self, context: &MetalContext) -> VelocityField {
        let nx = self.grid.nx() as usize;
        let ny = self.grid.ny() as usize;
        let nz = self.grid.nz() as usize;

        let grid = s_field::Grid4D::new(
            [nx, ny, nz, 1],
            [self.grid.dx(), self.grid.dy(), self.grid.dz(), 1.0],
        );

        //
        // GPU -> CPU
        //
        let u = self.u.download(context);
        let v = self.v.download(context);
        let w = self.w.download(context);

        //
        // Vec -> ScalarField
        //
        let mut u_field = s_field::ScalarField::<f32>::zeros(grid.clone());

        let mut v_field = s_field::ScalarField::<f32>::zeros(grid.clone());

        let mut w_field = s_field::ScalarField::<f32>::zeros(grid.clone());

        u_field.replace(u);
        v_field.replace(v);
        w_field.replace(w);

        VelocityField::from_fields(grid, u_field, v_field, w_field)
    }
}

use crate::{GpuVelocityField, GradientOperator, MetalContext, VelocityGradient};

pub struct VelocityGradientOperator {
    gradient: GradientOperator,
}

impl VelocityGradientOperator {
    pub fn new(gradient: GradientOperator) -> Self {
        Self { gradient }
    }

    pub fn compute(&self, context: &MetalContext, velocity: &GpuVelocityField) -> VelocityGradient {
        let u_gradient = self.gradient.gradient(context, &velocity.u);

        let v_gradient = self.gradient.gradient(context, &velocity.v);

        let w_gradient = self.gradient.gradient(context, &velocity.w);

        VelocityGradient {
            u_dx: u_gradient.dx,

            u_dy: u_gradient.dy,

            u_dz: u_gradient.dz,

            v_dx: v_gradient.dx,

            v_dy: v_gradient.dy,

            v_dz: v_gradient.dz,

            w_dx: w_gradient.dx,

            w_dy: w_gradient.dy,

            w_dz: w_gradient.dz,
        }
    }
}

/// 海流の移動速度を確認しながら、f32のバイナリデータを読み込む。
fn read_f32_file(path: &str) -> Result<Vec<f32>> {
    let bytes = fs::read(path).with_context(|| format!("ファイルを読み込めません: {path}"))?;

    anyhow::ensure!(
        bytes.len() % 4 == 0,
        "f32ファイルのサイズが4の倍数ではありません: {path}"
    );

    let values = bytes
        .chunks_exact(4)
        .map(|chunk| f32::from_le_bytes([chunk[0], chunk[1], chunk[2], chunk[3]]))
        .collect::<Vec<_>>();

    anyhow::ensure!(
        values.iter().all(|value| value.is_finite()),
        "{path} に非有限値があります"
    );

    Ok(values)
}

/// バイナリデータから海流の速度場を読み込む。
fn load_surface_velocity() -> Result<VelocityField> {
    // 配列の順序:
    // time × depth × latitude × longitude
    let grid = Grid4D::new(
        [1, DEPTH_COUNT, GRID_HEIGHT, GRID_WIDTH],
        [1.0, 1.0, 0.04, 0.08],
    );

    let latitudes = (0..GRID_HEIGHT)
        .map(|i| 34.52000045776367 + i as f64 * 0.04)
        .collect::<Vec<_>>();

    let longitudes = (0..GRID_WIDTH)
        .map(|i| 138.239990234375 + i as f64 * 0.08)
        .collect::<Vec<_>>();

    let depths = vec![0.0_f32, 10.0, 20.0, 50.0, 100.0, 200.0];
    let times = vec![0.0_f64];

    let coordinates = Coordinates::new(grid.clone(), latitudes, longitudes, depths, times);

    let u_path = format!("{RAW_DATA_DIR}/u.bin");
    let v_path = format!("{RAW_DATA_DIR}/v.bin");

    let u = read_f32_file(&u_path)?;
    let v = read_f32_file(&v_path)?;

    let expected = grid.len();

    anyhow::ensure!(
        u.len() == expected,
        "u.bin のデータ数が不正です: {} != {}",
        u.len(),
        expected
    );

    anyhow::ensure!(
        v.len() == expected,
        "v.bin のデータ数が不正です: {} != {}",
        v.len(),
        expected
    );

    let mut velocity = VelocityField::with_coordinates(grid, coordinates);

    velocity.u_mut().replace(u);
    velocity.v_mut().replace(v);
    velocity.w_mut().fill(0.05);
    println!(
        "Ocean velocity loaded: {} x {} grid, {} depth levels",
        GRID_WIDTH, GRID_HEIGHT, DEPTH_COUNT
    );

    // 速度データの統計を表示する。
    //
    // Fieldのデータ取得APIはプロジェクトの実装に依存するため、
    // ここでは読み込んだバイナリを直接再利用して統計を計算する。
    let u = read_f32_file(&u_path)?;
    let v = read_f32_file(&v_path)?;

    let mut min_speed = f32::INFINITY;
    let mut max_speed = 0.0_f32;
    let mut moving_points = 0usize;

    for (&u_value, &v_value) in u.iter().zip(v.iter()) {
        let speed = u_value.hypot(v_value);

        min_speed = min_speed.min(speed);
        max_speed = max_speed.max(speed);

        if speed >= 0.02 {
            moving_points += 1;
        }
    }

    println!(
        "Current speed range: {:.4} - {:.4} m/s",
        min_speed, max_speed
    );
    println!("Moving grid points: {}", moving_points);

    Ok(velocity)
}

/// 海流の開始位置を格子から生成する。
fn generate_seeds(coordinates: &Coordinates, bounds: &GeoBounds) -> Vec<[f32; 3]> {
    let mut seeds = Vec::new();

    for y in (0..coordinates.height()).step_by(2) {
        for x in 0..coordinates.width() {
            let Some(lon) = coordinates.longitude_at(x) else {
                continue;
            };

            let Some(lat) = coordinates.latitude_at(y) else {
                continue;
            };

            if lon < bounds.lon_min
                || lon > bounds.lon_max
                || lat < bounds.lat_min
                || lat > bounds.lat_max
            {
                continue;
            }

            seeds.push([lon as f32, lat as f32, 100.0]);
        }
    }

    seeds
}

use crate::coordinate::{GeoBounds, GeoConverter};
use crate::flow::FlowVertex;
use crate::heightmap::HeightMap;

/// 海底地形データから等深線を生成する。
///
/// # Arguments
///
/// * `heightmap` - 海底地形の標高データ（海底は負の値）
/// * `bounds` - 地理座標の範囲
/// * `vertical_scale` - 地形表示時の垂直方向の倍率
/// * `interval_m` - 等深線の間隔（メートル）
///
/// # Returns
///
/// 等深線を構成する線分の頂点配列。
pub fn generate_bathymetry_contours(
    heightmap: &HeightMap,
    bounds: GeoBounds,
    vertical_scale: f32,
    interval_m: f32,
) -> Vec<FlowVertex> {
    let width = heightmap.width;
    let height = heightmap.height;

    if width < 2 || height < 2 || interval_m <= 0.0 {
        return Vec::new();
    }

    let converter = GeoConverter::new(bounds, 100_000.0);
    let mut vertices = Vec::new();

    // 250 mなど、指定された間隔ごとに等深線を生成する。
    for z in 0..height - 1 {
        let lat0 =
            bounds.lat_min + (bounds.lat_max - bounds.lat_min) * z as f64 / (height - 1) as f64;

        let lat1 = bounds.lat_min
            + (bounds.lat_max - bounds.lat_min) * (z + 1) as f64 / (height - 1) as f64;

        for x in 0..width - 1 {
            let lon0 =
                bounds.lon_min + (bounds.lon_max - bounds.lon_min) * x as f64 / (width - 1) as f64;

            let lon1 = bounds.lon_min
                + (bounds.lon_max - bounds.lon_min) * (x + 1) as f64 / (width - 1) as f64;

            // セルの四隅。
            let corners = [
                (lon0, lat0, heightmap.get(x, z)),
                (lon1, lat0, heightmap.get(x + 1, z)),
                (lon1, lat1, heightmap.get(x + 1, z + 1)),
                (lon0, lat1, heightmap.get(x, z + 1)),
            ];

            // 標高データが不正なセルはスキップする。
            if corners.iter().any(|(_, _, h)| !h.is_finite()) {
                continue;
            }

            let min_height = corners
                .iter()
                .map(|(_, _, h)| *h)
                .fold(f32::INFINITY, f32::min);

            let max_height = corners
                .iter()
                .map(|(_, _, h)| *h)
                .fold(f32::NEG_INFINITY, f32::max);

            // 海底を含まないセルには等深線を生成しない。
            if min_height >= 0.0 {
                continue;
            }

            // 対角線で四角形を2つの三角形に分割する。
            let triangles = [[0_usize, 1_usize, 2_usize], [0_usize, 2_usize, 3_usize]];

            let first_level = (min_height / interval_m).ceil() as i32;
            let last_level = (max_height / interval_m).floor() as i32;

            for level_index in first_level..=last_level {
                let target = level_index as f32 * interval_m;

                // 海底の等深線だけを描画する。
                if target >= 0.0 {
                    continue;
                }

                for triangle in triangles {
                    let mut intersections: Vec<(f64, f64)> = Vec::with_capacity(3);

                    for [a_idx, b_idx] in [[0_usize, 1_usize], [1, 2], [2, 0]] {
                        let a = corners[triangle[a_idx]];
                        let b = corners[triangle[b_idx]];

                        let h_a = a.2;
                        let h_b = b.2;

                        // 等深線と交差する辺を判定する。
                        let crosses =
                            (h_a < target && h_b >= target) || (h_b < target && h_a >= target);

                        if !crosses {
                            continue;
                        }

                        let denominator = h_b - h_a;

                        if denominator.abs() < f32::EPSILON {
                            continue;
                        }

                        let t = ((target - h_a) / denominator) as f64;

                        let lon = a.0 + (b.0 - a.0) * t;
                        let lat = a.1 + (b.1 - a.1) * t;

                        intersections.push((lon, lat));
                    }

                    // 三角形と等深線の交点が2つなら線分を作る。
                    if intersections.len() != 2 {
                        continue;
                    }

                    let (lon_a, lat_a) = intersections[0];
                    let (lon_b, lat_b) = intersections[1];

                    let mut world_a = converter.geo_to_world(lon_a, lat_a);
                    let mut world_b = converter.geo_to_world(lon_b, lat_b);

                    // 地形表面との重なりを避けるため、わずかに下げる。
                    let y = target * vertical_scale + 0.5;
                    world_a[1] = y;
                    world_b[1] = y;

                    vertices.push(FlowVertex { position: world_a });

                    vertices.push(FlowVertex { position: world_b });
                }
            }
        }
    }

    vertices
}

use crate::{coordinate::GeoConverter, flow::FlowVertex};

use s_field::VelocityField;

/// 静的な流線を生成する。
pub struct FlowGenerator;

impl FlowGenerator {
    pub fn generate(
        velocity: &VelocityField,
        converter: &GeoConverter,
        seeds: &[[f32; 3]],
        dt_seconds: f32,
        length: usize,
        time: f64,
    ) -> Vec<FlowVertex> {
        let mut vertices = Vec::new();

        let Some(coordinates) = velocity.coordinates() else {
            return vertices;
        };

        let bounds = converter.bounds();

        for seed in seeds {
            let mut longitude = seed[0] as f64;
            let mut latitude = seed[1] as f64;
            let depth = seed[2];

            for _ in 0..length {
                if !coordinates.contains(longitude, latitude, depth) {
                    break;
                }

                if longitude < bounds.lon_min
                    || longitude > bounds.lon_max
                    || latitude < bounds.lat_min
                    || latitude > bounds.lat_max
                {
                    break;
                }

                let Some(vel) = velocity.sample(longitude, latitude, depth, time) else {
                    break;
                };

                let u = vel[0] as f64;
                let v = vel[1] as f64;

                if !u.is_finite() || !v.is_finite() || u.hypot(v) < 0.02 {
                    break;
                }

                let east_m = u * dt_seconds as f64;
                let north_m = v * dt_seconds as f64;

                let delta_lat = north_m / 110_540.0;

                let cos_lat = latitude.to_radians().cos();

                if cos_lat.abs() < 1.0e-6 {
                    break;
                }

                let delta_lon = east_m / (111_320.0 * cos_lat);

                let next_longitude = longitude + delta_lon;
                let next_latitude = latitude + delta_lat;

                if !coordinates.contains(next_longitude, next_latitude, depth) {
                    break;
                }

                if next_longitude < bounds.lon_min
                    || next_longitude > bounds.lon_max
                    || next_latitude < bounds.lat_min
                    || next_latitude > bounds.lat_max
                {
                    break;
                }

                let mut world_start = converter.ocean_to_world(longitude, latitude, depth);

                let mut world_end = converter.ocean_to_world(next_longitude, next_latitude, depth);

                // 既存の海流表示と同じ高さ補正。
                world_start[1] += 100.0;
                world_end[1] += 100.0;

                vertices.push(FlowVertex {
                    position: world_start,
                });

                vertices.push(FlowVertex {
                    position: world_end,
                });

                longitude = next_longitude;
                latitude = next_latitude;
            }
        }

        vertices
    }
}

/// 1つの流速粒子と、その軌跡を保持する。
struct FlowParticle {
    origin: [f64; 2],
    origin_depth: f32,

    longitude: f64,
    latitude: f64,
    depth: f32,

    trail: Vec<[f32; 3]>,
}

/// 流速データを使って粒子を時間発展させるシステム。
pub struct FlowParticleSystem {
    particles: Vec<FlowParticle>,
    max_trail_points: usize,
}

impl FlowParticleSystem {
    /// 初期位置から粒子を作成する。
    ///
    /// seedsは[経度、緯度、深度]。
    pub fn new(seeds: &[[f32; 3]], max_trail_points: usize) -> Self {
        let particles = seeds
            .iter()
            .map(|seed| FlowParticle {
                origin: [seed[0] as f64, seed[1] as f64],
                origin_depth: seed[2],

                longitude: seed[0] as f64,
                latitude: seed[1] as f64,
                depth: seed[2],

                trail: Vec::new(),
            })
            .collect();

        Self {
            particles,
            max_trail_points: max_trail_points.max(2),
        }
    }

    /// 全粒子を1フレーム分だけ移動させる。
    pub fn update(
        &mut self,
        velocity: &VelocityField,
        converter: &GeoConverter,
        dt_seconds: f32,
        speed_multiplier: f32,
        time: f64,
    ) {
        let Some(coordinates) = velocity.coordinates() else {
            return;
        };

        let bounds = converter.bounds();

        let dt = dt_seconds.clamp(0.0, 0.05) * speed_multiplier.max(0.0) * 100.0;

        if !dt.is_finite() || dt <= 0.0 {
            return;
        }

        for particle in &mut self.particles {
            let position_is_valid =
                coordinates.contains(particle.longitude, particle.latitude, particle.depth)
                    && particle.longitude >= bounds.lon_min
                    && particle.longitude <= bounds.lon_max
                    && particle.latitude >= bounds.lat_min
                    && particle.latitude <= bounds.lat_max;

            if !position_is_valid {
                Self::reset_particle(particle);
                continue;
            }

            let Some(vel) = velocity.sample_interpolated(
                particle.longitude,
                particle.latitude,
                particle.depth,
                time,
            ) else {
                Self::reset_particle(particle);
                continue;
            };

            let u = vel[0] as f64;
            let v = vel[1] as f64;
            let w = vel[2] as f64;

            if !u.is_finite() || !v.is_finite() || !w.is_finite() {
                Self::reset_particle(particle);
                continue;
            }

            if u.hypot(v).hypot(w) < 0.02 {
                continue;
            }

            let east_m = u * dt as f64;
            let north_m = v * dt as f64;

            let delta_lat = north_m / 110_540.0;

            let cos_lat = particle.latitude.to_radians().cos();

            if cos_lat.abs() < 1.0e-6 {
                Self::reset_particle(particle);
                continue;
            }

            let delta_lon = east_m / (111_320.0 * cos_lat);

            let next_longitude = particle.longitude + delta_lon;

            let next_latitude = particle.latitude + delta_lat;

            // 深度は海面0、下向きを正とする。
            let next_depth = particle.depth - (w as f32) * dt;
            let next_position_is_valid = next_depth.is_finite()
                && coordinates.contains(next_longitude, next_latitude, next_depth)
                && next_longitude >= bounds.lon_min
                && next_longitude <= bounds.lon_max
                && next_latitude >= bounds.lat_min
                && next_latitude <= bounds.lat_max;

            if !next_position_is_valid {
                Self::reset_particle(particle);
                continue;
            }

            let mut world_start =
                converter.ocean_to_world(particle.longitude, particle.latitude, particle.depth);

            let mut world_end = converter.ocean_to_world(next_longitude, next_latitude, next_depth);

            // 既存の海流表示と同じ高さ補正。
            world_start[1] += 100.0;
            world_end[1] += 100.0;

            if particle.trail.is_empty() {
                particle.trail.push(world_start);
            }

            particle.trail.push(world_end);

            if particle.trail.len() > self.max_trail_points {
                let excess = particle.trail.len() - self.max_trail_points;

                particle.trail.drain(0..excess);
            }

            particle.longitude = next_longitude;
            particle.latitude = next_latitude;
            particle.depth = next_depth;
        }
    }

    /// 粒子を初期位置に戻し、古い軌跡を消す。
    fn reset_particle(particle: &mut FlowParticle) {
        particle.longitude = particle.origin[0];
        particle.latitude = particle.origin[1];
        particle.depth = particle.origin_depth;

        particle.trail.clear();
    }

use wgpu::util::DeviceExt;

use crate::flow::FlowVertex;

/// GPU上に保持する潮流線メッシュ。
pub struct GpuFlowMesh {
    vertex_buffer: wgpu::Buffer,
    vertex_count: u32,
    capacity_bytes: u64,
}

impl GpuFlowMesh {
    /// 指定した最大頂点数でGPUバッファを確保する。
    pub fn with_capacity(device: &wgpu::Device, max_vertices: usize) -> Self {
        let capacity_vertices = max_vertices.max(1);

        let capacity_bytes = (capacity_vertices * std::mem::size_of::<FlowVertex>()) as u64;

        let vertex_buffer = device.create_buffer(&wgpu::BufferDescriptor {
            label: Some("dynamic_flow_vertex_buffer"),
            size: capacity_bytes,
            usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
            mapped_at_creation: false,
        });

        Self {
            vertex_buffer,
            vertex_count: 0,
            capacity_bytes,
        }
    }

    /// 初期頂点データからGPUメッシュを作成する。
    pub fn from_flow_vertices(device: &wgpu::Device, vertices: &[FlowVertex]) -> Self {
        let capacity_vertices = vertices.len().max(1);

        let capacity_bytes = (capacity_vertices * std::mem::size_of::<FlowVertex>()) as u64;

        let initial_contents = if vertices.is_empty() {
            bytemuck::cast_slice(&[FlowVertex {
                position: [0.0, 0.0, 0.0],
            }])
        } else {
            bytemuck::cast_slice(vertices)
        };

        let vertex_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
            label: Some("flow_vertex_buffer"),
            contents: initial_contents,
            usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
        });

        Self {
            vertex_buffer,
            vertex_count: vertices.len() as u32,
            capacity_bytes,
        }
    }

    /// CPU側で更新した頂点データをGPUへ転送する。
    ///
    /// バッファの容量を超えた場合は false を返す。
    pub fn update_vertices(&mut self, queue: &wgpu::Queue, vertices: &[FlowVertex]) -> bool {
        let required_bytes = (vertices.len() * std::mem::size_of::<FlowVertex>()) as u64;

        if required_bytes > self.capacity_bytes {
            return false;
        }

        if !vertices.is_empty() {
            queue.write_buffer(&self.vertex_buffer, 0, bytemuck::cast_slice(vertices));
        }

        self.vertex_count = vertices.len() as u32;

        true
    }

    pub fn vertex_buffer(&self) -> &wgpu::Buffer {
        &self.vertex_buffer
    }

    pub fn vertex_count(&self) -> u32 {
        self.vertex_count
    }
}

/// 海流の流線を描画するGPUパイプライン。
pub struct FlowPipeline {
    pipeline: wgpu::RenderPipeline,
}

impl FlowPipeline {
    /// 海流の流線用パイプラインを作成する。
    pub fn new(
        device: &wgpu::Device,
        format: wgpu::TextureFormat,
        shader: &wgpu::ShaderModule,
        camera_layout: &wgpu::BindGroupLayout,
    ) -> Result<Self> {
        let layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
            label: Some("flow_pipeline_layout"),
            bind_group_layouts: &[camera_layout],
            push_constant_ranges: &[],
        });

        let pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
            label: Some("flow_pipeline"),
            layout: Some(&layout),

            vertex: wgpu::VertexState {
                module: shader,
                entry_point: Some("vs_main"),
                compilation_options: wgpu::PipelineCompilationOptions::default(),
                buffers: &[FlowVertex::layout()],
            },

            fragment: Some(wgpu::FragmentState {
                module: shader,
                entry_point: Some("fs_main"),
                compilation_options: wgpu::PipelineCompilationOptions::default(),
                targets: &[Some(wgpu::ColorTargetState {
                    format,
                    blend: Some(wgpu::BlendState::ALPHA_BLENDING),
                    write_mask: wgpu::ColorWrites::ALL,
                })],
            }),

            primitive: wgpu::PrimitiveState {
                topology: wgpu::PrimitiveTopology::LineList,
                ..Default::default()
            },

            depth_stencil: Some(wgpu::DepthStencilState {
                format: wgpu::TextureFormat::Depth32Float,
                depth_write_enabled: false,
                depth_compare: wgpu::CompareFunction::LessEqual,
                stencil: wgpu::StencilState::default(),
                bias: wgpu::DepthBiasState::default(),
            }),

            multisample: wgpu::MultisampleState::default(),
            multiview: None,
            cache: None,
        });

        Ok(Self { pipeline })
    }

    /// 作成済みの描画パイプラインを取得する。
    pub fn pipeline(&self) -> &wgpu::RenderPipeline {
        &self.pipeline
    }
}

pub struct VelocityMesh {
    pub vertices: Vec<VelocityVertex>,
    pub indices: Vec<u32>,
}

impl VelocityMesh {
    pub fn from_field(field: &VelocityField, stride: usize) -> Self {
        let shape = field.shape();

        let nx = shape[2];
        let ny = shape[1];
        let nz = shape[0];

        let mut vertices = Vec::new();
        let mut indices = Vec::new();

        let mut id = 0u32;

        for z in (0..nz).step_by(stride) {
            for y in (0..ny).step_by(stride) {
                for x in (0..nx).step_by(stride) {
                    let u = *field.u().get(0, z, y, x).unwrap_or(&0.0);

                    let v = *field.v().get(0, z, y, x).unwrap_or(&0.0);

                    let w = *field.w().get(0, z, y, x).unwrap_or(&0.0);

                    let scale = 20.0;

                    vertices.push(VelocityVertex {
                        position: [x as f32, y as f32, -(z as f32)],
                        velocity: [u, v, w],
                    });

                    vertices.push(VelocityVertex {
                        position: [
                            x as f32 + u * scale,
                            y as f32 + v * scale,
                            -(z as f32) + w * scale,
                        ],
                        velocity: [u, v, w],
                    });

                    indices.push(id);
                    indices.push(id + 1);

                    id += 2;
                }
            }
        }

        Self { vertices, indices }
    }
}

use s_field::{
    BoundaryField, Coordinates, Field4D, Grid4D, OceanField, SalinityField, TemperatureField,
    VelocityField,
};

use s_io::SioReader;

use crate::error::Result;

pub struct BoundaryReader {
    reader: SioReader,
}

impl BoundaryReader {
    pub fn open(path: &str) -> Result<Self> {
        let reader = SioReader::open(path)?;

        Ok(Self { reader })
    }

    pub fn read(&mut self) -> Result<OceanField> {
        //
        // Metadata
        //

        let sio_metadata = self.reader.metadata().clone();

        let mut metadata = s_field::Metadata::new();

        metadata.set_dataset_name(sio_metadata.name);

        metadata.set_title("Suruga Ocean Format Boundary Dataset");

        metadata.set_comment(sio_metadata.description);

        metadata.insert_attribute("sof_version", sio_metadata.version);

        //
        // Grid
        //

        let size = self.reader.read_dataset("grid_size")?;

        let spacing = self.reader.read_dataset("grid_spacing")?;

        let grid = Grid4D::new(
            [
                size[0] as usize,
                size[1] as usize,
                size[2] as usize,
                size[3] as usize,
            ],
            [spacing[0], spacing[1], spacing[2], spacing[3]],
        );

        //
        // Coordinates
        //

        let latitude = self
            .reader
            .read_dataset("latitude")?
            .into_iter()
            .map(|v| v as f64)
            .collect();

        let longitude = self
            .reader
            .read_dataset("longitude")?
            .into_iter()
            .map(|v| v as f64)
            .collect();

        let depth = self.reader.read_dataset("depth")?;

        let time = self
            .reader
            .read_dataset("time")?
            .into_iter()
            .map(|v| v as f64)
            .collect();

        let coordinates = Coordinates::new(grid, latitude, longitude, depth, time);

        //
        // Velocity
        //

        let u = self.reader.read_dataset("velocity_u")?;

        let v = self.reader.read_dataset("velocity_v")?;

        let w = self.reader.read_dataset("velocity_w")?;

        let mut velocity = VelocityField::new(*coordinates.grid());

        velocity.u_mut().replace(u);

        velocity.v_mut().replace(v);

        velocity.w_mut().replace(w);

        velocity.set_coordinates(coordinates.clone());
        //
        // Temperature
        //

        let temperature = self.reader.read_dataset("temperature")?;

        let temperature =
            TemperatureField::from_field(Field4D::from_vec(*coordinates.grid(), temperature));

        //
        // Salinity
        //

        let salinity = self.reader.read_dataset("salinity")?;

        let salinity = SalinityField::from_field(Field4D::from_vec(*coordinates.grid(), salinity));
        let boundary = BoundaryField::zeros(*coordinates.grid());
        //
        // OceanField
        //

        Ok(OceanField::new(
            coordinates,
            metadata,
            velocity,
            temperature,
            salinity,
            boundary,
        ))
    }
}

use std::path::Path;

use s_field::{Coordinates, Grid4D, VelocityField};

use s_io::{Metadata, SioReader, SioWriter};

use crate::error::Result;

// ==================================================
// Velocity Writer
// ==================================================

pub struct VelocityWriter {
    writer: SioWriter,
}

impl VelocityWriter {
    pub fn create<P: AsRef<Path>>(path: P) -> Result<Self> {
        Ok(Self {
            writer: SioWriter::create(path.as_ref().to_str().unwrap())?,
        })
    }

    pub fn write(&mut self, velocity: &VelocityField) -> Result<()> {
        let metadata = Metadata {
            name: "Velocity Field".into(),
            version: "0.1.0".into(),
            description: "Ocean velocity field (u,v,w)".into(),
        };

        self.writer.write_metadata(&metadata)?;

        let size = velocity.grid().size();

        self.writer.write_array(
            "grid_size",
            &[
                size[0] as f32,
                size[1] as f32,
                size[2] as f32,
                size[3] as f32,
            ],
        )?;

        let spacing = velocity.grid().spacing();

        self.writer.write_array("grid_spacing", &spacing)?;

        self.writer.write_array("u", velocity.u().as_slice())?;

        self.writer.write_array("v", velocity.v().as_slice())?;

        self.writer.write_array("w", velocity.w().as_slice())?;

        Ok(())
    }

    pub fn finish(self) -> Result<()> {
        self.writer.finish()?;

        Ok(())
    }
}

// ==================================================
// Velocity Reader
// ==================================================

pub struct VelocityReader {
    reader: SioReader,
}

impl VelocityReader {
    pub fn open<P: AsRef<Path>>(path: P) -> Result<Self> {
        Ok(Self {
            reader: SioReader::open(path.as_ref().to_str().unwrap())?,
        })
    }

    pub fn read(&mut self, coordinates: Coordinates) -> Result<VelocityField> {
        let size = self.read_size()?;

        let spacing = self.read_spacing()?;

        let grid = Grid4D::new(size, spacing);

        let u = self.reader.read_array("u")?;

        let v = self.reader.read_array("v")?;

        let w = self.reader.read_array("w")?;

        let mut velocity = VelocityField::new(grid.clone());

        velocity.u_mut().replace(u);

        velocity.v_mut().replace(v);

        velocity.w_mut().replace(w);

        velocity.set_coordinates(coordinates);

        Ok(velocity)
    }

    fn read_size(&mut self) -> Result<[usize; 4]> {
        let v = self.reader.read_array("grid_size")?;

        Ok([v[0] as usize, v[1] as usize, v[2] as usize, v[3] as usize])
    }

    fn read_spacing(&mut self) -> Result<[f32; 4]> {
        let v = self.reader.read_array("grid_spacing")?;

        Ok([v[0], v[1], v[2], v[3]])
    }
}

use anyhow::Result;

use std::sync::Arc;

use s_field::{
    BoundaryField, Coordinates, Grid4D, Metadata, OceanField, SalinityField, TemperatureField,
    VelocityField,
};

use s_metal::{MetalContext, OceanSolver};

use simulation::Simulation;

use s_io::SioWriter;

fn main() -> Result<()> {
    pollster::block_on(run())
}

async fn run() -> Result<()> {
    println!("=== Suruga Ocean Simulator ===");

    //
    // Ocean field initialization
    //

    let grid = Grid4D::new([1, 32, 64, 64], [1.0, 10.0, 0.01, 0.01]);

    let coordinates = Coordinates::new(
        grid.clone(),
        vec![0.0; 64],
        vec![0.0; 64],
        vec![0.0; 32],
        vec![0.0; 1],
    );

    let ocean = OceanField::new(
        coordinates,
        Metadata::new(),
        VelocityField::new(grid.clone()),
        TemperatureField::new(grid.clone()),
        SalinityField::new(grid.clone()),
        BoundaryField::new(
            VelocityField::new(grid.clone()),
            VelocityField::new(grid.clone()),
            VelocityField::new(grid.clone()),
            VelocityField::new(grid.clone()),
        ),
    );

    //
    // Metal simulation
    //

    let context = Arc::new(MetalContext::new());

    let solver = OceanSolver::new(context.clone());

    let mut simulation = Simulation::new(&context, ocean, solver);

    //
    // Time integration
    //

    let steps = 100;
    let dt = 0.1;

    for i in 0..steps {
        simulation.step(dt);

        println!("step={} time={}", i, simulation.time);
    }

    //
    // Download velocity from GPU
    //

    println!("Downloading velocity field...");

    let velocity = simulation.state.download_velocity(&context);

    //
    // Save result
    //

    println!("Writing velocity output...");

    let mut writer =
        SioWriter::create("suruga_velocity.bin").expect("failed to create velocity file");

    writer
        .write_array("velocity_u", velocity.u().as_slice())
        .expect("failed to write velocity_u");

    writer
        .write_array("velocity_v", velocity.v().as_slice())
        .expect("failed to write velocity_v");

    writer
        .write_array("velocity_w", velocity.w().as_slice())
        .expect("failed to write velocity_w");

    writer.finish().expect("failed to finish velocity file");

    println!("Simulation finished.");

    Ok(())
}
